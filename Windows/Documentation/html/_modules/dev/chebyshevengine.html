    def cheb_points(N):
        """
        Check if number is not greater than one return error.

        Parameters
        ----------
        N : int
            it must be greater than 1.

        Returns
        -------
        value : val
            It returns a value.

        """
        if int(N) > 1:
            return np.cos((np.arange(N + 1) * np.pi) / N)
        else:
            print("message: Cheb Point {} must be a Greater than 1 ".format(N))



[docs]    @staticmethod
    def parametrisation(a, b, t):
        """
        Parametrisation used to linearly transform interval [0,1] to [a,b].

        Parameters
        ----------
        a : int
            Range interval value a (a, b)

        b : int
            Range interval value b (a, b)

        t : int
            it must be greater than 1.

        Returns
        -------
        value : val
            It returns a value.

        """

        if (0 <= t).all() and (t <= 1).all():
            value = t * b + (1 - t) * a
            return value
        else:
            raise Exception

    def cheb_pts_interval(N, a, b):
        """
        Check if entry a in interval is less than entry b in interval, else return error.

        Parameters
        ----------
        N : int
            No of cheb points to generate.

        a : int
            Range interval value a (a, b)

        b : int
            Range interval value b (a, b)

        Returns
        -------
        anchor_pts : int
            It returns anchor points.

        Raises
        ------
        BoundaryError : outside interval
             Interval boundary point a must be a less than boundary pointinterval b.

        """

        if a < b:
            cheb_pts = ChebyshevEngine.cheb_points(N)
            t = (abs(cheb_pts - 1)) / 2
            anchor_pts = ChebyshevEngine.parametrisation(b, a, t)
            return anchor_pts
        else:
            print(
                'Message error: Interval boundary point {} must be a less than boundary pointinterval {} '.format(a, b))
