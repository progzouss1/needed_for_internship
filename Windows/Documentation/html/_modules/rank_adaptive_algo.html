[docs]class RankAdaptiveAlgo(object):
    """
        Class that manages the run of the  Rank Adaptive Algorithm

        The class initialiser takes in the domain over which we build a Chebyshev Tensor in TT format and the number of Chebyshev points per dimension.

        To build the Chebyshev Tensor in TT format we need values on a subgrid of the Chebyshev grid. The subgrid can be randly generated by the class instance or may be set by the user.

        The subgrid must then be evaluated by the function to be approximated and stored within the class instance

        Once the subgrid and values on them are stored within the class instance, a training and validation set can be generated

        The Rank Adaptive algorithm can then be run. This generates the Chebyshev Tensor in TT format that approximates the function of interest.

        For more details on the algorithms implemented here, we refer to

        Steinlechner, M. Riemannian optimization for high-dimensional tensor completion. SIAM J. Sci. Comput. 38.  (2016). S461–S484.

        Glau, K. Kressner, F. Statti, F. Low-rank tensor approximation for Chebyshev interpolation in parametric option pricing. (2019) https://arxiv.org/abs/1902.04367


    """

    def __init__(self, dimension, num_cheb_pts, intervals):
        """
        Parameters
        ----------
        dimension : int
             Number of variables

        num_cheb_pts : list
             Number of Chebyshev points per dimension. Right now the number of Chebyshev points must be the same for all dimensions

        intervals : list of lists of floats
             List containing lists with two elements. Each list with two elements gives the range of a variable

        Raises
        ------
        ValueError
            dimension needs to be greater than 0
        ValueError
            dimension must be greater than 0
        ValueError
            When length of num_cheb_pts does not match dimension
        TypeError : type mismatch
            Entries in num_cheb_pts must all be integers
        ValueError
            All entries in num_cheb_pts must have the same value
        ValueError
            When intervals are not of length 2
        ValueError
            When number of intervals is not the same as dimension
        """

        self.dimension = dimension
        self.num_cheb_pts = num_cheb_pts
        self.intervals = intervals  # Only used for the generation of Cheb points per dimension
        self.__cheb_pts_tensor()

[docs]    def subgrid_by_number(self, n):
        """
        Randomly sample n grid points. This creates the subgrid used to build the Chebyshev Tensor in TT format

        Subgrid is randomly generated. Subgrid points may be repeated in this sampling. Repeats are removed. Therefore, user should be aware that the number of subgrid points returned may be lower than the number requested..


        Parameters
        ----------
        n : int
             value greater than 0.

        Returns
        -------
        subgrid_points_unique : numpy array
             2  dimensional numpy array containing the subrid of the tensor for which we need values to run the rank adaptive algorithm

        Raises
        ------
        TypeError : type mismatch
             Parameter n must be an integer

        ValueError
             Parameter n must be greater than 1

        """
        if not isinstance(n, int):
            raise TypeError("Subrid must have at least two points. Only {} were asked for".format(n))
        elif n < 1:
            # if n and n > 1:
            raise ValueError("The parameter must be greater than 1")
        else:
            subgrid_points_unique = self.__sample_points_calculation(n)
            return subgrid_points_unique



[docs]    def subgrid_by_percent(self, p):
        """
        Randomly sample a subgrid of the full Chebyshev grid by specifying a percentage p. This creates the subgrid used to build the Chebyshev Tensor in TT format

        Subgrid is randomly generated. Subgrid points may be repeated in this sampling. Repeats are removed. Therefore, user should be aware that the number of subgrid points returned may be lower than the number requested..


        Parameters
        ----------
        p : float
             percentage between 0 and 1.

        Returns
        -------
        subgrid_points_unique : numpy array
             2  dimensional numpy array containing the subrid of the tensor for which we need values to run the rank adaptive algorithm

        Raises
        ------
        ValueError
            Parameter p must be between zero and one
        """

        if not (0 <= p <= 1):
            raise ValueError("Parameter p must be between 0 and 1")
        else:
            tensor_size = self.get_tensor_size()
            percentage = int(round(p * tensor_size))
            subgrid_points_unique = self.__sample_points_calculation(percentage)
            return subgrid_points_unique


    def __sample_points_calculation(self, n):
        """
        This method does the work for subgrid_by_number and subgrid_by_percent.
        It does the random sampling and removes repeats

        Parameters
        ----------
        n : int
             value greater than 0.

        Returns
        -------
        subgrid_points_unique : numpy array
             2  dimensional numpy array containing the subrid of the tensor for which we need values to run the rank adaptive algorithm

        """
        subgrid_entries = np.empty((n, self.dimension), dtype=int)
        subgrid_points = np.empty((n, self.dimension))
        for m, pt in enumerate(self.cheb_pts_per_dimension):
            rand_entries = np.random.choice(range(len(pt)), n)
            subgrid_entries[:, m] = rand_entries
            subgrid_points[:, m] = pt[rand_entries]

        _, indices = np.unique(subgrid_entries, axis=0, return_index=True)
        subgrid_entries_unique = subgrid_entries[np.sort(indices)]
        subgrid_points_unique = subgrid_points[np.sort(indices)]
        self.subgrid_entries = subgrid_entries_unique

        return subgrid_points_unique


[docs]    def set_subgrid(self, subgrid_points):
        """
        This method lets the user manually set subgrid.

        Parameters
        ----------
        subgrid_points : numpy array
             array of random cheb_pts. Note that these are not the entries representing the subgrid but the actual values.

        Raises
        ------
        ValueError : 
             subgrid_points must be 2D numpy array

        ValueError : NaN check
             subgrid_points must not contain NaN values

        ValueError : size mismatch
             subgrid_points must have as many columns as dimension specified for tensor (dimension is an attribute of class intance)

        """
        subgrid_points = np.array(subgrid_points)
        if subgrid_points.ndim == 2:
            if np.any(~np.isnan(subgrid_points)):
                if subgrid_points.shape[1] == self.dimension:
                    subgrid_entries = np.empty(subgrid_points.shape)
                    for m, pts in enumerate(subgrid_points):
                        for n, entry in enumerate(pts):
                            cheb_pts = self.cheb_pts_per_dimension[n]
                            if round(entry) in np.round(self.cheb_pts_per_dimension[n]):
                                idx = list(cheb_pts).index(entry)
                                subgrid_entries[m][n] = idx
                            else:
                                raise ValueError("Entry {} at location {} is not present in cheb "
                                                 "points.".format(entry, n))
                    self.subgrid_entries = subgrid_entries
                else:
                    raise ValueError("Subgrid_points columns must be equal to dimension.")
            else:
                raise ValueError("Subgrid_points must not contain NaN values.")
        else:
            raise ValueError("Subgrid_points should be 2 Dimensional Numpy Array.")



[docs]    def set_subgrid_values(self, values):
        """
        Stores the values corresponding the subgrid within the class instance.

        Parameters
        ----------
        values : numpy array 1D
             values for train and validation.

        Raises
        ------
        ValueError : dimension of values parameter
             values parameter should be a 1D numpy array

        AttributeError :
             subgrid_entries must already exist within the class for this method to be called

        ValueError : length check
             Length of values must be the same as length of subgrid_entries

        ValueError : NaN check
             Values param musst not contain NaN values
        """
        if np.ndim(values) == 1:
            if hasattr(self, "subgrid_entries"):
                if np.shape(self.subgrid_entries)[0] == len(values):
                    if not np.isnan(np.array(values)).any():
                        self.subgrid_vals = np.array(values)
                    else:
                        raise ValueError('Values must not contain NaN values.')
                else:
                    raise ValueError(
                        "Length of values: {}, must be equal to rows of subgrid_entries: {}.".format(len(values), self.subgrid_entries.shape[0]))
            else:
                raise AttributeError(
                    "To set values of a subgrid, the attribute subgrid_entries must exist already. First, call subgrid_by_number, subgrid_by_percent or set_subgrid, and then set corresponding values using set_subgrid_values")
        else:
            raise ValueError("The parameter values must be a 1D numpy array")



[docs]    def gen_train_val_data(self, **kwargs):
        """
        Function splits subgrid_entries and subgrid values into training and validation,If no split percentage is passed, function assumes 0.8


        Parameters
        ----------
        percentage : float
             percentage value for splitting tensor values.

        Raises
        ------
        ValueError : outside of range
             Percentage should be between 0 and 1.

        AttributeError :
             There is no data to generate training and validating data sets.First, call subgrid_by_number, subgrid_by_percent or set_subgrid, and set corresponding values using set_subgrid_values

        """
        if hasattr(self, "subgrid_vals") and hasattr(self, "subgrid_entries"):
            percentage = kwargs.get('percentage', 0.8)
            if 0 < percentage < 1:

                split_pct = int(percentage * len(self.subgrid_vals))
                self.subgrid_entries_train, self.subgrid_entries_val = \
                    self.subgrid_entries[:split_pct, :], self.subgrid_entries[split_pct:, :]

                split_val = int(percentage * len(self.subgrid_vals))
                self.subgrid_vals_train = self.subgrid_vals[:split_val]
                self.subgrid_vals_val = self.subgrid_vals[split_val:]

            else:
                raise ValueError("Percentage:{} should be between 0 and 1".format(percentage))
        else:
            raise AttributeError(
                "There is no data to generate training and validating data sets. First, call subgrid_by_number, subgrid_by_percent or set_subgrid, and set corresponding values using set_subgrid_values")

    def __gen_sorted_subgrid_train_ptr(self):

        """
        Generates gen_subgrid_train_ptr used in alternating least squares scheme
        sorted_entries_train_ptrs  is a dictonary of ctype arrays with integers
        It is important to generated it before the als routine is run to speed the code
        """

        self.sorted_entries_train_ptrs = {}
        for core_idx in range(self.dimension):
            sorted_tensor_entries_train, sorted_values_train, __ = AlternatingLeastSquares.sort_tensor_pts(
                self.subgrid_entries_train, self.subgrid_vals_train, core_idx)

            ## -- Indices
            sorted_entries = np.array(sorted_tensor_entries_train.tolist(), dtype=np.intc)
            sorted_entries_ptr = sorted_entries.ctypes.data_as(ctypes.POINTER(ctypes.c_int * sorted_entries.size))
            self.sorted_entries_train_ptrs[core_idx] = sorted_entries_ptr

[docs]    def run_completion_algo(self, tensor, **kwargs):
        """
        This method runs the completion algorithm based an the Alternating Least Squares.

        Parameters
        ----------
        tensor : TT tensor
            To be modified by the algorithm

        **kwargs : dictionary with algorithm parameters
            tolerance: float
                Accuracy of approximation we want to reach (relative error)
                Default set to 1e^-3
            rel_tolerance: float
                If the algorithm improves the accuracy less than re_tolerance, the algo stops
                Default set to 1e^-8
            max_iters: int
                Number of iterations for completion algo based on ALS scheme
            print_progress: boolean value
                If true it prints the progress of the algorithm


        Returns
        -------
        tensor : TT tensor
            it is a tt tensor (sciki_tt).

        train_error : numpy array 1D
             Array with training errors 

        validation_error : numpy array 1D
             Array with validation errors 


        Raises
        ------
        ValueError : 
             tolerance, rel_tolerance, max_iters have to be greater than 0
        TypeError : type mismatch
             max_iters has to be an integer
        TypeError : type mismatch
             print_progress has to be a boolean value
        """

        tolerance, rel_tolerance, max_iters, print_progress = self.__param_check_run_completion(**kwargs)

        train_error = -100 * np.ones(2 * max_iters)
        validation_error = -100 * np.ones(2 * max_iters)

        norm_subgrid_entries_train = np.linalg.norm(self.subgrid_vals_train)
        norm_subgrid_entries_val = np.linalg.norm(self.subgrid_vals_val)

        # if tensor.ranks[5] == 8:
        #     print(f"Shape core before orth inside completion algo: {tensor.cores[4].shape}\n")

        tensor = TTChebyshevUtils.orthogonalize(tensor, 0)

        # if tensor.ranks[5] == 8:
        #     print(f"Shape core after orth inside completion algo: {tensor.cores[4].shape}\n")

        for i in range(max_iters):

            # print('Forward sweep', '\n')
            # t0 = time.time()
            for core_idx in range(self.dimension - 1):
                # print(core_idx)
                core_mod = AlternatingLeastSquares.alternating_least_squares(
                    self.subgrid_entries_train, self.subgrid_vals_train, tensor, core_idx, self.num_cheb_pts_ptr,
                    self.sorted_entries_train_ptrs[core_idx])

                tensor.cores[core_idx] = core_mod  # place core_mod inside of tensor
                tensor = TTChebyshevUtils.orth_left(tensor, core_idx)

            ## -- Error collection for forward sweep
            err_train = self.__tensor_error(self.subgrid_entries_train,
                                            self.subgrid_vals_train, tensor)
            train_error[2 * i] = np.sqrt(2 * err_train) / norm_subgrid_entries_train

            err_val = self.__tensor_error(self.subgrid_entries_val,
                                          self.subgrid_vals_val, tensor)
            validation_error[2 * i] = np.sqrt(2 * err_val) / norm_subgrid_entries_val

            ## -- Error check to see if we exit algorithm
            if train_error[2 * i] < tolerance and validation_error[2 * i] < tolerance:
                if print_progress:
                    print("\n")
                    print("\tConverged at iteration: {}".format(i))
                    print("\tTrain error: {:.3e}".format(train_error[2 * i]))
                    print("\tValidation error {:.3e}".format(validation_error[2 * i]))
                    print("\tLeaving algo\n")

                train_error = train_error[0:2 * i + 1]
                validation_error = validation_error[0:2 * i + 1]
                return tensor, train_error, validation_error

            if i > 0:

                ## -- Stagnation check to see if we exit algorithm
                rel_tol = abs(train_error[2 * i] - train_error[2 * i - 1]) / train_error[2 * i]
                rel_tol_test = abs(validation_error[2 * i] - validation_error[2 * i - 1]) / \
                               validation_error[2 * i]

                if (rel_tol < rel_tolerance) and (rel_tol_test < rel_tolerance):
                    if print_progress:
                        print("\n")
                        print("\tAt iteration: {}".format(i))
                        print("\tRelative error change training: {:.3e}".format(rel_tol))
                        print("\tRelative error change testing: {:.3e}".format(rel_tol_test))
                        print("\tLeaving algo\n")

                    train_error = train_error[:2 * i + 1]
                    validation_error = validation_error[:2 * i + 1]
                    return tensor, train_error, validation_error

            # print('Backward sweep', '\n')
            for core_idx in range(self.dimension - 1, 0, -1):
                # print(core_idx)

                core_mod = AlternatingLeastSquares.alternating_least_squares(
                    self.subgrid_entries_train,
                    self.subgrid_vals_train, tensor, core_idx, self.num_cheb_pts_ptr,
                    self.sorted_entries_train_ptrs[core_idx])

                tensor.cores[core_idx] = core_mod  # place core_mod inside of tensor
                tensor = TTChebyshevUtils.orth_right(tensor, core_idx)

            ## -- Error collection for backward sweep
            error_train = self.__tensor_error(self.subgrid_entries_train,
                                              self.subgrid_vals_train, tensor)
            train_error[2 * i + 1] = np.sqrt(2 * error_train) / norm_subgrid_entries_train

            error_val = self.__tensor_error(self.subgrid_entries_val,
                                            self.subgrid_vals_val, tensor)
            validation_error[2 * i + 1] = np.sqrt(2 * error_val) / norm_subgrid_entries_val

            ## -- Error check to see if we exit algorithm
            if train_error[2 * i + 1] < tolerance and validation_error[2 * i + 1] < tolerance:
                if print_progress:
                    print("\n")
                    print("\tConverged at iteration: {}".format(i))
                    print("\tTrain error: {:.3e}".format(train_error[2 * i]))
                    print("\tValidation error {:.3e}".format(validation_error[2 * i]))
                    print("\tLeaving algo\n")

                train_error = train_error[:2 * i + 2]
                validation_error = validation_error[:2 * i + 2]
                return tensor, train_error, validation_error

            if i > 1:
                ## -- Stagnation check to see if we exit algorithm
                rel_tol = abs(train_error[2 * i + 1] - train_error[(2 * i + 1) - 1]) / \
                          train_error[2 * i + 1]
                rel_tol_test = abs(
                    validation_error[2 * i + 1] - validation_error[(2 * i + 1) - 1]) / \
                               validation_error[2 * i + 1]

                if (rel_tol < rel_tolerance) and (rel_tol_test < rel_tolerance):
                    if print_progress:
                        print("\n")
                        print("\tAt iteration: {}".format(i))
                        print("\tRelative error change training: {:.3e}".format(rel_tol))
                        print("\tRelative error change testing: {:.3e}".format(rel_tol_test))
                        print("\tLeaving algo\n")

                    train_error = train_error[:2 * i + 2]
                    validation_error = validation_error[:2 * i + 2]
                    return tensor, train_error, validation_error

        return tensor, train_error, validation_error

    @staticmethod
    def __increase_rank(tensor, idx, epsilon):
        """
        Increases the rank of the tensor at the core given by index idx

        Parameters
        ----------
        tensor : TT
             Chebyshev Tensor in TT format.

        idx : int
             Core for which we increase rank
             Index is integer and less than the dimension of the tensor.

        epsilon : float
             Used to randomly sample values and extend rank of core

        Returns
        -------
        tensor : TT
             Chebyshev Tensor in TT format.

        Raises
        ------
        ValueError : outside interval
             The index passed is outside the accepted range. It must be greater than zero and less than the dimension of the tensor

        """

        if idx > 0 and idx < tensor.order:

            r = tensor.ranks
            n = tensor.row_dims
            tensor_new = copy.deepcopy(tensor)

            tensor_new.cores[idx - 1] = np.concatenate((tensor.cores[idx - 1],
                                                        epsilon * np.random.randn(r[idx - 1],
                                                                                  n[idx - 1], 1,
                                                                                  1)), axis=3)
            tensor_new.cores[idx] = np.concatenate((tensor.cores[idx],
                                                    epsilon * np.random.randn(1, n[idx], 1,
                                                                              r[idx + 1])), axis=0)
            tensor_new.ranks[idx] = tensor.ranks[idx] + 1

            if idx == 5 and tensor.ranks[idx] == 7:
                print(f"shape of old core at idx - 1: {tensor.cores[idx - 1].shape}")
                print(f"shape of new core at idx - 1: {tensor_new.cores[idx - 1].shape}")

                print(f"shape of old core at idx: {tensor.cores[idx].shape}")
                print(f"shape of new core at idx: {tensor_new.cores[idx].shape}")

            return tensor_new
        else:

            raise ValueError(
                "The index passed {} is outside the accepted range. It must be greater than zero and less than the dimension of the tensor".format(
                    idx))

        """
        if isinstance(tensor, TT):

            c_tensor = TTChebyshevUtils.c_tensor(tensor, self.num_cheb_pts)

            if pts.ndim == 2:

                num_pts = pts.shape[0]
                vals = np.empty((num_pts))

                for n in range(num_pts):
                    pt = pts[n]
                    pt_scaled = np.empty(self.dimension)

                    for m in range(self.dimension):
                        pt_1d = pt[m]
                        if pt_1d < self.intervals[m][0] or pt_1d > self.intervals[m][1]:
                            raise ValueError("Point {} outside the"
                                            " domain given by [{}, {}]".format(pt_1d,
                                                                               self.intervals[m][0],
                                                                               self.intervals[m][1]))
                        else:
                            pt_scaled[m] = TTChebyshevUtils.domain_scaler(self.intervals[m][0],
                                                                          self.intervals[m][1], pt_1d)
                    val = self.cheb_tensor_eval(c_tensor, pt_scaled)
                    vals[n] = val
                return vals
            else:
                raise ValueError("Parameter pts must be 2 Dimensional Numpy Array.")
        else:
            raise TypeError("The tensor must be of type TT.")
